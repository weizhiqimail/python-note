# 1. urllib的用法
+ urllib是Python3.x中提供的一系列操作URL的库，可以模拟用户使用浏览器访问网页
+ 使用步骤
    - `from urllib import request`
    - 请求URL`response = request.urlopen('http://www.baidu.com')`
    - 使用相应对象输出数据`print(response.read().decode('utf-8'))`

+ 使用get
```python
# 基本用法
from urllib import request

response = request.urlopen('http://www.baidu.com')

print(response.read().decode('utf-8'))
```

```python
# 携带User-Agent头
req = request.Request('https://www.baidu.com')

req.add_header('User-Agent', 'Mozilla/5.0 (Windows NT 10.0; …) Gecko/20100101 Firefox/61.0')
res = request.urlopen(req)

print(res.read().decode('utf-8'))
```
+ 使用post，`from urllib import parse`
    - 使用`urlencode`生成`post`数据
    - `request.urlopen(req, data=postData.encode('utf-8'))`
    - 得到请求状态`res.status`
    - 得到服务器类型`res.reason`

```python
from urllib.request import urlopen
from urllib.request import Request
from urllib import parse

req = Request('http://www.thsrc.com.tw/tw/TimeTable/SearchResult')
postData = parse.urlencode([
    ('StartStation', '2f940836-cedc-41ef-8e28-c2336ac8fe68'),
    ('EndStation', 'fbd828d8-b1da-4b06-a3bd-680cdca4d2cd'),
    ('SearchDate', '2018/07/24'),
    ('SearchTime', '18:00'),
    ('SearchWay', 'DepartureInMandarin')
])

req.add_header('User-Agent', 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/67.0.3396.99 Safari/537.36')

resp = urlopen(req, data=postData.encode('utf-8'))

print(resp.read().decode('utf-8'))
```

# 2. BeautifulSoup
+ 安装`pip install bs4`
+ 使用例子

```python
from bs4 import BeautifulSoup
import re

html_doc = """
<html><head><title>The Dormouse's story</title></head>
<body>
<p class="title"><b>The Dormouse's story</b></p>

<p class="story">Once upon a time there were three little sisters; and their names were
<a href="http://example.com/elsie" class="sister" id="link1">Elsie</a>,
<a href="http://example.com/lacie" class="sister" id="link2">Lacie</a> and
<a href="http://example.com/tillie" class="sister" id="link3">Tillie</a>;
and they lived at the bottom of a well.</p>

<p class="story">...</p>
"""

soup = BeautifulSoup(html_doc, 'html.parser')

print(soup.prettify())

print(soup.title.string)

print(soup.find(id='link2').string)
print(soup.find(id='link2').get_text())

for link in soup.find_all('a'):
    print(link.string)

print(soup.find('p', {'class': 'story'}).get_text())

for tag in soup.find_all(re.compile('^b')):
    print(tag.name)

da = soup.findAll('a', href=re.compile(r'^http://example\.com/'))

print(da)
```
+ 爬取维基百科的一个例子

```python
# -*- coding:utf-8 -*-

from urllib.request import urlopen
from bs4 import BeautifulSoup as bs
import re

#  获取链接地址信息
resp = urlopen('https://en.wikipedia.org/wiki/English_Wikipedia').read().decode('utf-8')

# 使用bs去解析页面
soup = bs(resp, 'html.parser')

# 获取所有以/wiki/开头的a标签的href属性
listUrls = soup.findAll('a', href=re.compile('^/wiki/'))

# 获取a链接中不是图片链接的href属性值
for url in listUrls:
    if not re.search("\.(jpg|JPG|jpeg|JPEG|png|PNG|gif|GIF)$", url['href']):
        # 输出所有的词条的内容，以及对应的url地址
        print(url.get_text(), '-------->', 'https://en.wikipedia.org' + url['href'])
```
