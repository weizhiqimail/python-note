## 1. 如何在列表，字典，集合中根据条件筛选数据
+ 列表

```python
# 列表
from random import randint

data = [randint(-10, 10) for x in range(10)]
res = filter(lambda x: x >= 0, data)
# print(list(res))

# 列表解析的方式，速度更快
res = [x for x in data if x >= 0]
# print(res)
```

+ 字典

```python
from random import randint

# 字典解析
d = {x: randint(60, 100) for x in range(1, 21)}
res = {k: v for k, v in d.items() if v >= 90}
print(res)
```

+ 集合

```python
from random import randint

# 集合解析
data = [randint(-10, 10) for x in range(10)]
s = set(data)
res = {x for x in s if x % 3 == 0}
print(res)
```

## 4. 如何根据字典中值的大小，对字典的项进行排序
+ 使用内置函数sorted

```python
from random import randint

data = {x: randint(60, 100) for x in 'xyzabc'}
# 直接对字典使用sorted，那么就会以字典的key的ascii码进行排序
print(sorted(data))
print(list(iter(data)))

# 得到data所有的key
keys = data.keys()

# 得到data所有的value
values = data.values()

# 把元组构成列表
res = list(zip(values, keys))

# 对列表进行排序
print(sorted(res))
```

+ 传递sorted函数的key参数

```python
data = {x: randint(60, 100) for x in 'xyzabc'}
# 传递sorted函数的key参数
print(sorted(data.items(), key=lambda x: x[1]))
```
## 5. 如何快速找到多个字典中的公共键(key)

+ 使用最原始的方法循环操作

```python
from random import randint, sample

user = 'abcdefg'

r1 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r2 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r3 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r4 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
print(r1)
print(r2)
print(r3)
print(r4)

res = []

for k in r1:
    if k in r2 and k in r3 and k in r4:
        res.append(k)

print(res)
```

+ 使用集合的操作

```python
from random import randint, sample

user = 'abcdefg'

r1 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r2 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r3 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r4 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}

print(r1.keys() & r2.keys() & r3.keys() & r4.keys())
```
+ 使用map和reduce函数

```python
from random import randint, sample
from functools import reduce

user = 'abcdefg'

r1 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r2 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r3 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}
r4 = {x: randint(1, 4) for x in sample(user, randint(3, 6))}

map(dict.keys, [r1, r2, r3, r4])

print(reduce(lambda a, b: a & b, map(dict.keys, [r1, r2, r3, r4])))
```
## 6. 如何让字典保持有序
+ 使用OrderedDict

```python
from collections import OrderedDict

d = OrderedDict()
d['marry'] = 10
d['jack'] = 12
d['sherry'] = 11
d['tom'] = 9

for x in d:
    print(x)
```

+ 示例

```python
from collections import OrderedDict
from time import time
from random import randint

d = OrderedDict()
players = list('ABCDEFGH')
start = time()
for x in range(8):
    input()
    p = players.pop(randint(0, 7 - x))
    end = time()
    print(x, x + 1, p, end - start)
    d[p] = (x + 1, end - start)

for k in d:
    print(k, d[k])
```

## 7. 如何实现用户的历史记录功能

+ 使用容量为n的队列存储历史记录
+ 使用标准库collections中的deque，是一个双端循环队列
+ 程序退出前，可以用pickle将队列对象存入文件，再次运行程序时将其导入

```python
from random import randint
from collections import deque
# import pickle

N = randint(0, 100)
history = deque([], 5)
# pickle.dump(history, open('history', 'w'))


def guess(k):
    if k == N:
        print('yes, you are right!')
    if k < N:
        print('%s is less than N' % k)
    else:
        print('%s is bigger than N' % k)
    return False


while True:
    num = input('please input a number: ')
    if num.isdigit():
        k = int(num)
        history.append(num)

        if guess(k):
            break
    elif num == 'history' or num == 'H':
        print(list(history))
```
